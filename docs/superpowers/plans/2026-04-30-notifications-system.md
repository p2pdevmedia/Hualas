# Notifications System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an in-app notification inbox plus Web Push (PWA) delivery, driven by 9 event types across activities, pickup notices, payments, capacity, and chat — with per-user preferences and a service module that each API route calls explicitly.

**Architecture:** A central `notification-service` exposes one function per event. After every relevant DB commit, the route fires-and-forgets a call to the service, which resolves recipients, applies preferences, inserts `Notification` rows in a single `createMany`, and dispatches Web Push via `web-push` in parallel with `Promise.allSettled`. Failures are logged and never block the originating operation. Frontend has a navbar bell + panel, a permission modal at first login, and a `/profile/notifications` preferences page.

**Tech Stack:** Next.js 14 App Router, TypeScript, Prisma, PostgreSQL, NextAuth, Zod, `web-push` library, Web Push API + Service Worker, Tailwind, Jest.

**Prerequisites:** spec at `docs/superpowers/specs/2026-04-30-notifications-system-design.md`. Read it first.

```ts
import { prisma } from '@/lib/prisma';
import type { NotificationType } from '@prisma/client';

export type Preference = { inApp: boolean; push: boolean };

const DEFAULT_PREFERENCE: Preference = { inApp: true, push: true };

export async function resolvePreferences(
  userIds: string[],
  type: NotificationType
): Promise<Map<string, Preference>> {
  const result = new Map<string, Preference>();
  if (userIds.length === 0) return result;

  const rows = await prisma.notificationPreference.findMany({
    where: { userId: { in: userIds }, type },
    select: { userId: true, inApp: true, push: true },
  });

  const byUser = new Map(rows.map((r) => [r.userId, { inApp: r.inApp, push: r.push }]));

  for (const userId of userIds) {
    result.set(userId, byUser.get(userId) ?? DEFAULT_PREFERENCE);
  }

  return result;
}
```

```ts
import webpush from 'web-push';

export type PushSubscriptionPayload = {
  endpoint: string;
  p256dh: string;
  auth: string;
};

export type PushPayload = {
  title: string;
  body: string;
  url?: string;
  data?: Record<string, unknown>;
};

export class PushDeliveryError extends Error {
  constructor(message: string, public statusCode: number) {
    super(message);
    this.name = 'PushDeliveryError';
  }
}

let configured = false;
function ensureConfigured() {
  if (configured) return;
  const subject = process.env.VAPID_SUBJECT;
  const publicKey = process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY;
  const privateKey = process.env.VAPID_PRIVATE_KEY;
  if (!subject || !publicKey || !privateKey) {
    throw new Error('VAPID env vars missing — see README');
  }
  webpush.setVapidDetails(subject, publicKey, privateKey);
  configured = true;
}

export async function sendPush(
  subscription: PushSubscriptionPayload,
  payload: PushPayload
): Promise<void> {
  ensureConfigured();
  try {
    await webpush.sendNotification(
      {
        endpoint: subscription.endpoint,
        keys: { p256dh: subscription.p256dh, auth: subscription.auth },
      },
      JSON.stringify(payload)
    );
  } catch (err) {
    const status = (err as { statusCode?: number }).statusCode ?? 0;
    throw new PushDeliveryError(
      `Push delivery failed (HTTP ${status})`,
      status
    );
  }
}
```

```ts
import { prisma } from '@/lib/prisma';

function unique<T>(arr: T[]): T[] {
  return Array.from(new Set(arr));
}

export async function recipientsForActivityDay(dayId: string): Promise<string[]> {
  const day = await prisma.activityDay.findUnique({
    where: { id: dayId },
    select: { activityId: true },
  });
  if (!day) return [];
  const participants = await prisma.activityParticipant.findMany({
    where: { activityId: day.activityId },
    select: { userId: true },
  });
  return unique(participants.map((p) => p.userId));
}

export async function recipientsForPickupNotice(noticeId: string): Promise<string[]> {
  const notice = await prisma.pickupNotice.findUnique({
    where: { id: noticeId },
    select: {
      activityDayId: true,
      alternatePersonUserId: true,
      activityDay: { select: { activityId: true } },
    },
  });
  if (!notice) return [];

  const dayProfs = await prisma.activityDayProfessor.findMany({
    where: { activityDayId: notice.activityDayId },
    select: { userId: true },
  });

  let profs = dayProfs.map((p) => p.userId);
  if (profs.length === 0) {
    const actProfs = await prisma.activityProfessor.findMany({
      where: { activityId: notice.activityDay.activityId },
      select: { userId: true },
    });
    profs = actProfs.map((p) => p.userId);
  }

  const all = [...profs];
  if (notice.alternatePersonUserId) all.push(notice.alternatePersonUserId);
  return unique(all);
}

export async function recipientsForPickupNoticeAcknowledged(
  ackId: string
): Promise<string[]> {
  const ack = await prisma.pickupNoticeAcknowledgment.findUnique({
    where: { id: ackId },
    select: { pickupNotice: { select: { createdById: true } } },
  });
  if (!ack) return [];
  return [ack.pickupNotice.createdById];
}

export async function recipientsForManualMovement(): Promise<string[]> {
  const users = await prisma.user.findMany({
    where: { isActive: true, role: { in: ['COUNTER', 'ADMIN', 'SUPER_ADMIN'] } },
    select: { id: true },
  });
  return users.map((u) => u.id);
}

export async function recipientsForCapacityFull(): Promise<string[]> {
  const users = await prisma.user.findMany({
    where: { isActive: true, role: { in: ['ADMIN', 'SUPER_ADMIN'] } },
    select: { id: true },
  });
  return users.map((u) => u.id);
}

export async function recipientsForActivityParticipantPayment(
  participantId: string
): Promise<string[]> {
  const p = await prisma.activityParticipant.findUnique({
    where: { id: participantId },
    select: { userId: true },
  });
  return p ? [p.userId] : [];
}

export async function recipientsForOrderPayment(
  orderId: string
): Promise<string[]> {
  const order = await prisma.order.findUnique({
    where: { id: orderId },
    select: { responsibleUserId: true },
  });
  return order?.responsibleUserId ? [order.responsibleUserId] : [];
}

export async function recipientsForChatMessage(
  messageId: string
): Promise<{ recipients: string[]; senderId: string; conversationId: string }> {
  const message = await prisma.message.findUnique({
    where: { id: messageId },
    select: {
      senderId: true,
      conversationId: true,
      conversation: {
        select: { participants: { select: { userId: true } } },
      },
    },
  });
  if (!message) {
    return { recipients: [], senderId: '', conversationId: '' };
  }
  const recipients = message.conversation.participants
    .map((p) => p.userId)
    .filter((id) => id !== message.senderId);
  return {
    recipients: unique(recipients),
    senderId: message.senderId,
    conversationId: message.conversationId,
  };
}
```

```ts
import { prisma } from '@/lib/prisma';
import type { NotificationType, Prisma } from '@prisma/client';
import { resolvePreferences } from './preferences';
import { sendPush, PushDeliveryError } from './push-sender';

export type DispatchInput = {
  type: NotificationType;
  recipients: string[];
  title: string;
  body: string;
  url?: string;
  data?: Prisma.InputJsonValue;
};

export async function dispatch(input: DispatchInput): Promise<void> {
  if (input.recipients.length === 0) return;

  const prefs = await resolvePreferences(input.recipients, input.type);

  let inAppRecipients = input.recipients.filter(
    (id) => prefs.get(id)?.inApp ?? true
  );

  // Type-specific filter: chat-message debounce.
  if (input.type === 'CHAT_MESSAGE_NEW') {
    const data = (input.data ?? {}) as { senderId?: string; conversationId?: string };
    if (data.senderId && data.conversationId) {
      const filtered: string[] = [];
      for (const userId of inAppRecipients) {
        const existing = await prisma.notification.findFirst({
          where: {
            userId,
            type: 'CHAT_MESSAGE_NEW',
            readAt: null,
            data: {
              path: ['senderId'],
              equals: data.senderId,
            },
            AND: {
              data: {
                path: ['conversationId'],
                equals: data.conversationId,
              },
            },
          },
          select: { id: true },
        });
        if (!existing) filtered.push(userId);
      }
      inAppRecipients = filtered;
    }
  }

  if (inAppRecipients.length > 0) {
    await prisma.notification.createMany({
      data: inAppRecipients.map((userId) => ({
        userId,
        type: input.type,
        title: input.title,
        body: input.body,
        url: input.url ?? null,
        data: input.data ?? Prisma.JsonNull,
      })),
    });
  }

  const pushRecipients = input.recipients.filter(
    (id) => prefs.get(id)?.push ?? true
  );
  if (pushRecipients.length === 0) return;

  const subs = await prisma.pushSubscription.findMany({
    where: { userId: { in: pushRecipients }, failedAt: null },
    select: { id: true, endpoint: true, p256dh: true, auth: true, userId: true },
  });

  const payload = {
    title: input.title,
    body: input.body,
    url: input.url,
    data: input.data,
  };

  const results = await Promise.allSettled(
    subs.map((sub) => sendPush(sub, payload))
  );

  const toFail: string[] = [];
  results.forEach((res, i) => {
    if (res.status === 'rejected') {
      const err = res.reason;
      if (
        err instanceof PushDeliveryError &&
        (err.statusCode === 404 || err.statusCode === 410)
      ) {
        toFail.push(subs[i].id);
      } else {
        console.error('[notifications] push failed', err);
      }
    }
  });

  if (toFail.length > 0) {
    await prisma.pushSubscription.updateMany({
      where: { id: { in: toFail } },
      data: { failedAt: new Date() },
    });
  }
}
```

```ts
export async function POST(req: Request) {
  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const body = await req.json().catch(() => null);
  const parsed = pushSubscriptionSchema.safeParse(body);
  if (!parsed.success) {

      userAgent,
      lastUsedAt: new Date(),

      endpoint: parsed.data.endpoint,
      p256dh: parsed.data.keys.p256dh,
      auth: parsed.data.keys.auth,
      userAgent,
```

```ts
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { prisma } from '@/lib/prisma';

export async function DELETE(
  _req: Request,
  { params }: { params: { id: string } }

  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
```

```ts
import { NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { prisma } from '@/lib/prisma';

export async function GET(req: Request) {
  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const url = new URL(req.url);
  const unreadOnly = url.searchParams.get('unread') === '1';
  const limit = Math.min(Number(url.searchParams.get('limit') ?? 20), 100);

  const where = unreadOnly
    ? { userId: session.user.id, readAt: null }
    : { userId: session.user.id };

  const [items, unreadCount] = await Promise.all([
    prisma.notification.findMany({
      where,
      orderBy: [{ readAt: 'asc' }, { createdAt: 'desc' }],
      take: limit,
      select: {
        id: true,
        type: true,
        title: true,
        body: true,
        url: true,
        readAt: true,
        createdAt: true,
      },
    }),
    prisma.notification.count({
      where: { userId: session.user.id, readAt: null },
    }),
  ]);

  return NextResponse.json({ items, unreadCount });
}

export async function PATCH() {
  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const result = await prisma.notification.updateMany({
    where: { userId: session.user.id, readAt: null },
    data: { readAt: new Date() },
  });
  return NextResponse.json({ updated: result.count });
```

```ts
import { NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { prisma } from '@/lib/prisma';
import { updatePreferenceSchema } from '@/lib/validations/notifications';
import type { NotificationType } from '@prisma/client';

const ALL_TYPES: NotificationType[] = [
  'ACTIVITY_DAY_NEW',
  'ACTIVITY_DAY_UPDATED',
  'PICKUP_NOTICE_CREATED',
  'PICKUP_NOTICE_ACKNOWLEDGED',
  'PAYMENT_MANUAL_CREATED',
  'PAYMENT_APPROVED',
  'PAYMENT_REJECTED',
  'ACTIVITY_CAPACITY_FULL',
  'CHAT_MESSAGE_NEW',
];

export async function GET() {
  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const rows = await prisma.notificationPreference.findMany({
    where: { userId: session.user.id },
    select: { type: true, inApp: true, push: true },
  });
  const map = new Map(rows.map((r) => [r.type, { inApp: r.inApp, push: r.push }]));
  const preferences: Record<string, { inApp: boolean; push: boolean }> = {};
  for (const t of ALL_TYPES) {
    preferences[t] = map.get(t) ?? { inApp: true, push: true };
  }
  return NextResponse.json({ preferences });
}

export async function PUT(req: Request) {
  const session = await getServerSession(authOptions);
  if (!session?.user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const body = await req.json().catch(() => null);
  const parsed = updatePreferenceSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ error: 'Invalid preference' }, { status: 400 });
  }
  await prisma.notificationPreference.upsert({
    where: { userId_type: { userId: session.user.id, type: parsed.data.type } },
    update: { inApp: parsed.data.inApp, push: parsed.data.push },
    create: {
      userId: session.user.id,
      type: parsed.data.type,
      inApp: parsed.data.inApp,
      push: parsed.data.push,
    },
  });
  return NextResponse.json({ ok: true });
}
```

```js
self.addEventListener('push', (event) => {
  if (!event.data) return;
  let payload;
  try {
    payload = event.data.json();
  } catch {
    payload = { title: 'Hualas', body: event.data.text() };
  }
  const { title, body, url, data } = payload;
  event.waitUntil(
    Promise.all([
      self.registration.showNotification(title || 'Hualas', {
        body: body || '',
        icon: '/favicon.ico',
        data: { url: url || '/', ...data },
      }),
      // Wake any open client so the bell can refresh.
      self.clients
        .matchAll({ type: 'window', includeUncontrolled: true })
        .then((clients) => {
          clients.forEach((c) => c.postMessage({ type: 'notification:new' }));
        }),
    ])
  );
});

self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  const url = (event.notification.data && event.notification.data.url) || '/';
  event.waitUntil(
    self.clients.matchAll({ type: 'window', includeUncontrolled: true }).then((clients) => {
      for (const client of clients) {
        if ('focus' in client) {
          client.focus();
          if ('navigate' in client) client.navigate(url);
          return;
        }
      }
      if (self.clients.openWindow) return self.clients.openWindow(url);
    })
  );
});
```

```tsx
'use client';

import { useEffect, useState } from 'react';
import { PushPermissionModal } from './push-permission-modal';

const DISMISS_KEY = 'pushPromptDismissedAt';
const DISMISS_WINDOW_MS = 7 * 24 * 60 * 60 * 1000;

function urlBase64ToUint8Array(base64String: string): Uint8Array {
  const padding = '='.repeat((4 - (base64String.length % 4)) % 4);
  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/');
  const raw = atob(base64);
  const out = new Uint8Array(raw.length);
  for (let i = 0; i < raw.length; i++) out[i] = raw.charCodeAt(i);
  return out;
}

function isIosWithoutPwa(): boolean {
  if (typeof window === 'undefined') return false;
  const isIos = /iPhone|iPad|iPod/.test(navigator.userAgent);
  const standalone = (window.navigator as unknown as { standalone?: boolean }).standalone === true;
  return isIos && !standalone;
}

async function registerAndSubscribe() {
  const reg = await navigator.serviceWorker.register('/sw.js');
  await navigator.serviceWorker.ready;
  const existing = await reg.pushManager.getSubscription();
  const sub =
    existing ??
    (await reg.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: urlBase64ToUint8Array(
        process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY ?? ''
      ),
    }));
  const json = sub.toJSON();
  await fetch('/api/notifications/subscriptions', {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({
      endpoint: json.endpoint,
      keys: { p256dh: json.keys?.p256dh, auth: json.keys?.auth },
    }),
  });
}

export function PushManager() {
  const [showModal, setShowModal] = useState(false);

  useEffect(() => {
    if (typeof window === 'undefined') return;
    if (!('serviceWorker' in navigator) || !('PushManager' in window)) return;

    navigator.serviceWorker.register('/sw.js').catch((err) => {
      console.error('[notifications] sw register failed', err);
    });

    const permission = Notification.permission;

    if (permission === 'granted') {
      registerAndSubscribe().catch((err) =>
        console.error('[notifications] subscribe failed', err)
      );
      return;
    }

    if (permission === 'default') {
      const dismissed = Number(localStorage.getItem(DISMISS_KEY) ?? 0);
      if (Date.now() - dismissed > DISMISS_WINDOW_MS) {
        setShowModal(true);
      }
    }
  }, []);

  const handleAccept = async () => {
    setShowModal(false);
    const result = await Notification.requestPermission();
    if (result === 'granted') {
      try {
        await registerAndSubscribe();
      } catch (err) {
        console.error('[notifications] subscribe failed', err);
      }
    } else {
      localStorage.setItem(DISMISS_KEY, String(Date.now()));
    }
  };

  const handleDismiss = () => {
    setShowModal(false);
    localStorage.setItem(DISMISS_KEY, String(Date.now()));
  };

  return (
    <PushPermissionModal
      open={showModal}
      onAccept={handleAccept}
      onDismiss={handleDismiss}
      isIosWithoutPwa={isIosWithoutPwa()}
    />
  );
}
```

```tsx
'use client';

import { useEffect, useState, useCallback } from 'react';
import { Bell } from 'lucide-react';
import { NotificationPanel } from './notification-panel';

type Item = {
  id: string;
  type: string;
  title: string;
  body: string;
  url: string | null;
  readAt: string | null;
  createdAt: string;
};

const POLL_MS = 60_000;

export function NotificationBell() {
  const [open, setOpen] = useState(false);
  const [items, setItems] = useState<Item[]>([]);
  const [unread, setUnread] = useState(0);

  const refresh = useCallback(async () => {
    try {
      const res = await fetch('/api/notifications?limit=10');
      if (!res.ok) return;
      const data = await res.json();
      setItems(data.items);
      setUnread(data.unreadCount);
    } catch (err) {
      console.error('[notifications] bell refresh failed', err);
    }
  }, []);

  useEffect(() => {
    refresh();
    const id = setInterval(refresh, POLL_MS);
    const onSwMessage = (event: MessageEvent) => {
      if (event.data?.type === 'notification:new') refresh();
    };
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.addEventListener('message', onSwMessage);
    }
    return () => {
      clearInterval(id);
      if ('serviceWorker' in navigator) {
        navigator.serviceWorker.removeEventListener('message', onSwMessage);
      }
    };
  }, [refresh]);

  const markRead = async (id: string) => {
    setItems((cur) => cur.map((i) => (i.id === id ? { ...i, readAt: new Date().toISOString() } : i)));
    setUnread((u) => Math.max(0, u - 1));
    await fetch(`/api/notifications/${id}`, { method: 'PATCH' });
  };

  const markAll = async () => {
    setUnread(0);
    setItems((cur) =>
      cur.map((i) => (i.readAt ? i : { ...i, readAt: new Date().toISOString() }))
    );
    await fetch('/api/notifications', { method: 'PATCH' });
  };

  return (
    <div className="relative">
      <button
        type="button"
        onClick={() => setOpen((v) => !v)}
        className="relative rounded p-2 hover:bg-gray-100"
        aria-label="Notificaciones"
      >
        <Bell className="h-5 w-5" />
        {unread > 0 && (
          <span className="absolute right-0 top-0 rounded-full bg-red-500 px-1 text-[10px] font-bold text-white">
            {unread > 9 ? '9+' : unread}
          </span>
        )}
      </button>
      {open && (
        <NotificationPanel items={items} onMarkAllRead={markAll} onMarkRead={markRead} />
      )}
    </div>
  );
}
```

```tsx
import type { NotificationType } from '@prisma/client';
import { TYPE_LABELS } from '@/lib/notifications/types-by-role';

type Prefs = Record<string, { inApp: boolean; push: boolean }>;

export function PreferencesForm({
  types,
  initial,
  subscriptions,

  types: NotificationType[];

      ...prefs[type],
```
